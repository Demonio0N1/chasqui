cmake_minimum_required(VERSION 3.16)
project(chasqui CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(CHASQUI_SINGLE_PRECISION "Build with float instead of double" OFF)
option(CHASQUI_NATIVE "Add -march=native on CPU builds" ON)
option(CHASQUI_ALLOW_FMA "Allow fused multiply-add contraction (faster, breaks bitwise mirror symmetry)" OFF)
option(CHASQUI_BUILD_TESTS "Build the symmetry unit test" ON)

# Kokkos: either an installed package (Kokkos_ROOT / CMAKE_PREFIX_PATH) or the
# submodule at external/kokkos
find_package(Kokkos QUIET)
if(NOT Kokkos_FOUND)
  if(EXISTS ${CMAKE_SOURCE_DIR}/external/kokkos/CMakeLists.txt)
    add_subdirectory(external/kokkos)
  else()
    message(FATAL_ERROR "Kokkos not found. Install it or run: git submodule update --init")
  endif()
endif()

add_executable(chasqui src/main.cpp)
target_include_directories(chasqui PRIVATE src)
target_link_libraries(chasqui PRIVATE Kokkos::kokkos)
if(CHASQUI_SINGLE_PRECISION)
  target_compile_definitions(chasqui PRIVATE CHASQUI_SINGLE_PRECISION)
endif()
if(CHASQUI_NATIVE AND NOT Kokkos_ENABLE_CUDA AND NOT Kokkos_ENABLE_HIP)
  target_compile_options(chasqui PRIVATE -march=native)
endif()
target_compile_options(chasqui PRIVATE -O3 -Wall -Wextra)
if(NOT CHASQUI_ALLOW_FMA)
  # a*b+c contracted to an FMA rounds differently from its mirror image a'*b'+c'; keeping
  # contraction off makes reflected flows bitwise reflected (tests/symmetry_test.cpp)
  target_compile_options(chasqui PRIVATE -ffp-contract=off)
endif()

if(CHASQUI_BUILD_TESTS)
  add_executable(symmetry_test tests/symmetry_test.cpp)
  target_include_directories(symmetry_test PRIVATE src)
  target_link_libraries(symmetry_test PRIVATE Kokkos::kokkos)
  target_compile_options(symmetry_test PRIVATE -O3)
  if(NOT CHASQUI_ALLOW_FMA)
    target_compile_options(symmetry_test PRIVATE -ffp-contract=off)
  endif()
  if(CHASQUI_NATIVE AND NOT Kokkos_ENABLE_CUDA AND NOT Kokkos_ENABLE_HIP)
    target_compile_options(symmetry_test PRIVATE -march=native)
  endif()
  if(CHASQUI_SINGLE_PRECISION)
    target_compile_definitions(symmetry_test PRIVATE CHASQUI_SINGLE_PRECISION)
  endif()
  add_executable(eigen_test tests/eigen_test.cpp)
  target_include_directories(eigen_test PRIVATE src)
  target_link_libraries(eigen_test PRIVATE Kokkos::kokkos)
  target_compile_options(eigen_test PRIVATE -O2)
  enable_testing()
  add_test(NAME symmetry COMMAND symmetry_test)
  add_test(NAME eigensystem COMMAND eigen_test)
endif()
